@section Scripts{
    <script>
        var apiUrl="http://localhost:5172/api/Student";

        function validationform()
        {
             var name = $("#name").val().trim();
             var email = $("#email").val().trim();
             var gender = $("input[name='gender']:checked").val();
             var dob = $("#dob").val();
             var course = $("#course").val();
             var hobbies = $("input[name=hobby]:checked");

             var message = "";
             var isValid = true;

             if(name ==="" || name.length <3)
             {
                 message +="Name must be 3<br>";
                 isValid = false;
             }

              if (email === "") 
              {
            message += "Email is required<br>";
            isValid = false;
             }
               if (!gender) {
              message += "Select gender<br>";
              isValid = false;
          }
               if (hobbies.length === 0) {
              message += "Select at least one hobby<br>";
              isValid = false;
          }

          if (dob === "") {
              message += "Select DOB<br>";
              isValid = false;
          }

          if (course === "") {
              message += "Select course<br>";
              isValid = false;
          }

          if(!isValid)
          {
              $("#message").html(message);

          }
          else{
              $("#message").html("");
          }

          return isValid;

        }

        function addstudent(){

              if(!validationform()) return;
        var hobbies = [];
        $("input[name='hobby']:checked").each(function(){
            hobbies.push($(this).val());
        });

        var student =
        {
            name: $("#name").val(),
            email:$("#email").val(),
            gender: $("input[name='gender']:checked").val(),
            hobbies:hobbies.join(","),
              dob: $("#dob").val(),
              course:$("#course").val()
        };

        $.ajax({
            url: apiUrl,
            type: "POST",
            contentType: "application/json",
            data: JSON.stringify(student),
            success: function(){
                alert("student add successfuly");
                window.location.href = "/Student/Index";
                
            },
            error: function(){
               alert("error");
            }
        })
      }
    </script>
}
