@section Scripts{
    <script>
        var apiUrl = "http://localhost:5172/api/Student";

        var id =  new URLSearchParams(window.location.search).get("id");


        function loadstudent(){

            $.ajax({
                url: apiUrl + "/" +id,
                type: "GET",

                success: function(s){
                    $("#name").val(s.name);
                    $("#email").val(s.email);
                    $("#dob").val(s.dob.split('T')[0]);
                    $("#course").val(s.course);

                    if(s.gender == "male")
                    {
                        $("input[value='male']").prop("checked", true)
                    }
                    else
                    {
                        $("input[value='female']").prop("checked",true)
                    }

                    if(s.hobbies.includes("reading"))
                    {
                        $("input[value='reading']").prop("checked",true)
                    }
                    if(s.hobbies.includes("sports"))
                    {
                        $("input[value='sports']").prop("checked",true)
                    }
                    if(s.hobbies.includes("music"))
                    {
                        $("input[value='music']").prop("checked",true)
                    }
                }
                
            });
        }

        function updatestudent()
        {
            var hobbies = [];

            $("input[name='hobby']:checked").each(function()
            {
                hobbies.push($(this).val());
            })

            var student = {
                name: $("#name").val(),
                email:$("#email").val(),
                gender: $("input[name='gender']:checked").val(),
                hobbies: hobbies.join(","),
                dob: $("#dob").val(),
                course: $("#course").val()
            };

            $.ajax({
                url: apiUrl + "/" + id,
                type: "PUT",
                contentType: "application/json",
                data: JSON.stringify(student),

                success: function(){
                    alert("update student automatically");
                    window.location.href = "/Student/Index";


                },

                error: function()
                {
                    alert("error");
                }
            });
        }

        $(document).ready(function (){
            loadstudent();
        })
    </script>    
}
